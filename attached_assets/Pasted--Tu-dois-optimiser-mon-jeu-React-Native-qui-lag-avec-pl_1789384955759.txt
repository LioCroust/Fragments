// ✅ 1. SPRITE ENNEMI MEMOIZÉ
const EnemySprite = React.memo(
  ({ enemy, enemyIndex, cell, spriteFrames }: any) => {
    const frame = useMemo(() => enemyFrameIndex(enemy), [enemy.phase]);
    const size = useMemo(() => enemyRenderSize(enemy.kind, cell, enemy.isMini), [enemy.kind, cell, enemy.isMini]);
    const motion = useMemo(() => enemyAnimationTransform(enemy, cell), [enemy.x, enemy.y, enemy.spin, cell]);
    const centerY = enemy.y + motion.offsetY;
    const rotationDegrees = motion.rotation * (180 / Math.PI);
    const transform = `translate(${enemy.x} ${centerY}) rotate(${rotationDegrees}) scale(${motion.scale})`;
    return <G key={`enemy-sprite-${enemyIndex}`} transform={transform}><SvgImage href={spriteFrames[enemy.kind][frame]} x={enemy.x - size.width / 2} y={enemy.y - size.height / 2} width={size.width} height={size.height} opacity={0.98} /></G>;
  },
  (prevProps: any, nextProps: any) => prevProps.enemy.x === nextProps.enemy.x && prevProps.enemy.y === nextProps.enemy.y && prevProps.enemy.phase === nextProps.enemy.phase && prevProps.enemy.spin === nextProps.enemy.spin && prevProps.cell === nextProps.cell && prevProps.enemy.kind === nextProps.enemy.kind && prevProps.enemy.isMini === nextProps.enemy.isMini
);

// ✅ 2. PROJECTILE MEMOIZÉ
const ProjectileSprite = React.memo(
  ({ projectile, projectileIndex, cell, sevenFireOrbSource }: any) => {
    const spriteSize = useMemo(() => sevenProjectileSize(cell), [cell]);
    return <SvgImage key={`projectile-${projectileIndex}`} href={sevenFireOrbSource} x={projectile.x - spriteSize / 2} y={projectile.y - spriteSize / 2} width={spriteSize} height={spriteSize} opacity={0.96} />;
  },
  (p: any, n: any) => p.projectile.x === n.projectile.x && p.projectile.y === n.projectile.y && p.cell === n.cell
);

// ✅ 3. MISSILE JOUEUR MEMOIZÉ
const PlayerMissileSprite = React.memo(
  ({ missile, missileIndex, cell, playerMissileSource, arenaBounds }: any) => {
    if (!pointInsidePerimeter(missile, arenaBounds)) return null;
    const spriteSize = useMemo(() => playerMissileSize(cell), [cell]);
    const rotationDegrees = useMemo(() => missile.angle * (180 / Math.PI), [missile.angle]);
    return <SvgImage key={`missile-${missileIndex}`} href={playerMissileSource} x={missile.x - spriteSize / 2} y={missile.y - spriteSize / 2} width={spriteSize} height={spriteSize} opacity={0.98} transform={`rotate(${rotationDegrees} ${missile.x} ${missile.y})`} />;
  },
  (p: any, n: any) => p.missile.x === n.missile.x && p.missile.y === n.missile.y && p.missile.angle === n.missile.angle && p.cell === n.cell
);

// ✅ 4. BOMBE MEMOIZÉE
const BombSprite = React.memo(
  ({ bomb, bombIndex, pickupSize, frame, coreReactorSpriteSource }: any) => {
    if (bomb.destroyed) return null;
    return <G key={`bomb-${bombIndex}`} transform={`translate(${bomb.x} ${bomb.y})`}><G clipPath="url(#bomb-clip)"><SvgImage href={coreReactorSpriteSource} x={-pickupSize / 2 - frame * pickupSize} y={-pickupSize / 2} width={pickupSize * CORE_REACTOR_SPRITE_FRAME_COUNT} height={pickupSize} opacity={0.98} /></G></G>;
  },
  (p: any, n: any) => p.bomb.x === n.bomb.x && p.bomb.y === n.bomb.y && p.bomb.destroyed === n.bomb.destroyed && p.frame === n.frame
);

// ✅ 5. DIAMANT MEMOIZÉ
const DiamondSprite = React.memo(
  ({ diamond, diamondIndex, cell, frame, pickupSize, diamondSpriteSource }: any) => {
    if (diamond.collected) return null;
    return <G key={`diamond-${diamondIndex}`} transform={`translate(${diamond.x} ${diamond.y})`}><G clipPath="url(#diamond-clip)"><SvgImage href={diamondSpriteSource} x={-pickupSize / 2 - frame * pickupSize} y={-pickupSize / 2} width={pickupSize * DIAMOND_SPRITE_FRAME_COUNT} height={pickupSize} opacity={0.94} /></G></G>;
  },
  (p: any, n: any) => p.diamond.x === n.diamond.x && p.diamond.y === n.diamond.y && p.diamond.collected === n.diamond.collected && p.frame === n.frame
);

// ✅ 6. SPIDER THREAD MEMOIZÉ
const SpiderThreadSprite = React.memo(
  ({ thread, threadIndex, cell, spiderWebSource }: any) => {
    const active = useMemo(() => spiderThreadIsActive(thread), [thread.remaining, thread.anchored]);
    const webSize = useMemo(() => cell * thread.webSizeCells, [cell, thread.webSizeCells]);
    const opacity = useMemo(() => active ? 0.82 : 0.92, [active]);
    return (
      <G key={`spider-thread-${threadIndex}`} opacity={opacity}>
        {!thread.anchored && (
          <>
            <Line x1={thread.start.x} y1={thread.start.y} x2={thread.end.x} y2={thread.end.y} stroke="#fff3d6" strokeWidth={2.8} strokeLinecap="round" />
            <Line x1={thread.start.x} y1={thread.start.y} x2={thread.end.x} y2={thread.end.y} stroke="#ff2bb5" strokeWidth={1.15} strokeLinecap="round" />
            <Line x1={thread.start.x} y1={thread.start.y} x2={thread.end.x} y2={thread.end.y} stroke="#00f3ff" strokeWidth={0.72} strokeLinecap="round" />
            <Circle cx={thread.end.x} cy={thread.end.y} r={3.2} fill="#fff3d6" />
          </>
        )}
        {thread.anchored && <SvgImage href={spiderWebSource} x={thread.target.x - webSize / 2} y={thread.target.y - webSize / 2} width={webSize} height={webSize} opacity={clamp(0.6 + thread.remaining * 0.08, 0.6, 0.88)} />}
      </G>
    );
  },
  (p: any, n: any) => p.thread.start.x === n.thread.start.x && p.thread.start.y === n.thread.start.y && p.thread.end.x === n.thread.end.x && p.thread.end.y === n.thread.end.y && p.thread.target.x === n.thread.target.x && p.thread.target.y === n.thread.target.y && p.thread.anchored === n.thread.anchored && p.thread.remaining === n.thread.remaining
);
