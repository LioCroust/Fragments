================================================================================
PROBLEMES IDENTIFIES
================================================================================

1. FLIP DES SPRITES
   - Transformations SVG/Skia appliquent scale(-1,1) ou (1,-1) au lieu de (1,1)
   - Matrice de rendu non normalisee
   - Les sprites apparaissent inverses

2. FPS BLOQUES A 30-40
   CAUSE: useSkiaImage() rappele a chaque render
   CAUSE: Render loop force state update inutile (setFrameCount)
   CAUSE: Canvas Skia re-painted a chaque frame meme sans changement
   CAUSE: Collisions recalculees meme HORS-ECRAN
   CAUSE: Aucun preload => images chargees pendant le gameplay

3. ASSETS NON PRECHARGES
   - Aucun systeme de preload au boot
   - Images chargees lazily => lag au 1er affichage
   - Multiple Image components re-render inutilement

```typescript
import { useEffect, useRef, useState } from 'react';
import { useImage as useSkiaImage } from '@shopify/react-native-skia';
import { Image as RNImage } from 'react-native';

type AssetCache = {
  images: Map<string, any>;
  sprites: Map<string, any>;
  loading: Set<string>;
  failed: Set<string>;
};

const assetCache: AssetCache = {
  images: new Map(),
  sprites: new Map(),
  loading: new Set(),
  failed: new Set(),
};

// Sprite sheet sources - A ADAPTER avec tes vraies sources
const SPRITE_SOURCES = {
  cutting: require('../assets/images/cutting-sprite-sheet.png'),
  shipSmoke: require('../assets/images/ship-smoke-sprite-sheet.png'),
  coreReactor: require('../assets/images/core-reactor-sprite-sheet.png'),
  diamond: require('../assets/images/neon-diamond-fragment-sprite-sheet.png'),
  sevenFireOrb: require('../assets/images/seven-fire-orb.png'),
  speedBoost: require('../assets/images/speed-boost-sprite.png'),
  spiderWeb: require('../assets/images/spider-web-destination.png'),
};

// Image sources - A ADAPTER avec tes vraies sources
const IMAGE_SOURCES = {
  loadingCover: require('../assets/images/loading-cover.png'),
  cockpitInterior: require('../assets/images/prism-warbird-interior-neon-console.png'),
  sector1: require('../assets/images/sector-1-space-background.png'),
  sector2: require('../assets/images/sector-2-space-background.png'),
  sector3: require('../assets/images/sector-3-space-background.png'),
  // Ajoute les 50 secteurs ici...
};

export const preloadAssets = async (): Promise<void> => {
  return new Promise((resolve) => {
    let loaded = 0;
    const total = Object.keys(SPRITE_SOURCES).length + Object.keys(IMAGE_SOURCES).length;

    const onLoad = () => {
      loaded++;
      if (loaded >= total) resolve();
    };

    // Preload sprites
    Object.entries(SPRITE_SOURCES).forEach(([key, source]) => {
      if (!assetCache.sprites.has(key)) {
        const image = new RNImage();
        image.onload = () => {
          assetCache.sprites.set(key, image);
          onLoad();
        };
        image.onerror = () => {
          assetCache.failed.add(key);
          onLoad();
        };
        assetCache.loading.add(key);
        image.src = source;
      }
    });

    // Preload images
    Object.entries(IMAGE_SOURCES).forEach(([key, source]) => {
      if (!assetCache.images.has(key)) {
        const image = new RNImage();
        image.onload = () => {
          assetCache.images.set(key, image);
          onLoad();
        };
        image.onerror = () => {
          assetCache.failed.add(key);
          onLoad();
        };
        assetCache.loading.add(key);
        image.src = source;
      }
    });
  });
};

export const usePreloadedImage = (key: string) => {
  return assetCache.images.get(key) || null;
};

export const usePreloadedSprite = (key: string) => {
  return assetCache.sprites.get(key) || null;
};

export const getAssetCacheStatus = () => ({
  imagesLoaded: assetCache.images.size,
  spritesLoaded: assetCache.sprites.size,
  failed: assetCache.failed.size,
});

export const useCachedSkiaImage = (source: any) => {
  const cache = useRef(new Map());
  const sourceKey = JSON.stringify(source);
  
  if (!cache.current.has(sourceKey)) {
    cache.current.set(sourceKey, useSkiaImage(source));
  }
  
  return cache.current.get(sourceKey);
};
```

SOLUTION 8: COLLISION DETECTION OPTIMIZATION (VIEWPORT CULLING)
================================================================================

Dans ton code de collision, ajoute:

const lastCollisionState = useRef(new Set());

const checkCollisionsOptimized = useCallback((entities: Entity[], screenWidth: number, screenHeight: number) => {
  const newCollisions = new Set();
  
  // ETAPE 1: Filter - Garder QUE les entities visibles (avec marge)
  const visibleEntities = entities.filter(e => 
    e.x > -100 && e.x < screenWidth + 100 &&
    e.y > -100 && e.y < screenHeight + 100
  );
  
  // ETAPE 2: Check - Collision check uniquement sur les visibles
  for (let i = 0; i < visibleEntities.length; i++) {
    for (let j = i + 1; j < visibleEntities.length; j++) {
      if (checkCollision(visibleEntities[i], visibleEntities[j])) {
        newCollisions.add(`${visibleEntities[i].id}-${visibleEntities[j].id}`);
      }
    }
  }
  
  // ETAPE 3: Diff - Tracker UNIQUEMENT les changements
  const changed = new Set([
    ...Array.from(newCollisions).filter(c => !lastCollisionState.current.has(c)),
    ...Array.from(lastCollisionState.current).filter(c => !newCollisions.has(c)),
  ]);
  
  lastCollisionState.current = newCollisions;
  return changed;
}, []);
